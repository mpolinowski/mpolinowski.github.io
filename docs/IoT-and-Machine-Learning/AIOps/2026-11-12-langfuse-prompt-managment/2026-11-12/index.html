LangFuse Prompt Management

Adding System Prompts​

Before I hardcoded a system prompt to a LangGraph agent. LangFuse allows us to run experiments with different prompts to record performance metrics - let's try to set this up! I start by copying the text from the ./prompts/system.txt file into LangFuse as a new prompt for the already created projects:

Now I need to remove the hardcoded system prompt from graph.py:

SYSTEM_PROMPT = (Path(__file__).parent / "prompts" / "system.txt").read_text(encoding="utf-8")

And retrieve the prompt from LangFuse instead - note that the generation has to be associated with the LangFuse prompt version to enable tracing:

from langfuse import get_client,  propagate_attributes

# load system prompt -> removed and replaced by dynamic prompt via Langfuse Prompt Management
# SYSTEM_PROMPT = (Path(__file__).parent / "prompts" / "system.txt").read_text(encoding="utf-8")
langfuse = get_client()


# define graph node
def chatbot(state: MessagesState):

    langfuse_prompt = langfuse.get_prompt(
        name="hello-world-system-prompt",
        type="text",
        label="production"
    )

    system_prompt = langfuse_prompt.get_langchain_prompt()

    messages = [
        SystemMessage(content=system_prompt),
        *state["messages"],
    ]

    # associate the generation with the LangFuse prompt version
    with propagate_attributes(prompt=langfuse_prompt):
        response = model.invoke(messages)

    return {"messages": [response]}

And just a few minor changes to the main function to actually use the versioned system prompt:

from langchain_core.messages import HumanMessage
from langfuse import propagate_attributes
from langfuse._client.get_client import get_client
from langfuse.langchain import CallbackHandler
from langgraph_hello_langfuse.config import langfuse_handler

from .graph import graph

langfuse = get_client()
langfuse_handler = CallbackHandler()

def main():
  session_id = str(uuid.uuid4())
  print("-------------------------------")
  print("LOGIN")
  username = input("\nUser Name: ")
  print(f"\nHi {username}! Your chat session ID is {session_id}")
  print("-------------------------------")

  while True:
    question = input(f"\nAsk your question: ")

    if question.lower() in ["exit", "quit"]:
      break

    with propagate_attributes(
        trace_name="LangGraph Hello World",
        user_id=username,
        session_id=session_id,
        tags=["hello-world", "ollama", "langgraph"],
        metadata={
        "model": "local-ollama",
        "environment": "development",
        },
    ):

        response = graph.invoke(
            {
            "messages": [
                HumanMessage(content=question)
            ]
            },
            config = {
                "callbacks": [langfuse_handler],
                "run_name": "chatbot-execution",
            },
        )

    ai_message = response["messages"][-1]

    usage = ai_message.usage_metadata or {}
    model = ai_message.response_metadata.get('model', 'Unknown model')
    input_tokens = usage.get("input_tokens", 0)
    output_tokens = usage.get("output_tokens", 0)
    total_tokens = usage.get("total_tokens", 0)

    print(f"\nAssistant: {ai_message.content}")

    # DEBUG
    print("----------------------------------")
    print(f"{model} token usage: {input_tokens}, {output_tokens}, {total_tokens}")
    print("----------------------------------\n")

  # send telemetry
  langfuse.flush()

A/B Testing​

To role out the new prompt in production you can run an A/B test:

import random

# do live A/B testing
prompt_label = random.choice(["production", "staging"]) 

# define graph node
def chatbot(state: MessagesState):

    langfuse_prompt = langfuse.get_prompt(
        name="hello-world-system-prompt",
        type="text",
        label=prompt_label
    )
