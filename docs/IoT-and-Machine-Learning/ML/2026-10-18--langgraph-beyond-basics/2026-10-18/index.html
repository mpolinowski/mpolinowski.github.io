Workflow :: Parallelization​

Running nodes in parallel in LangGraph is just connecting the START of your StateGraph to mor than one functional nodes - just like that:

from typing import TypedDict
from pydantic import SecretStr
from langchain_openai import ChatOpenAI
from langgraph.graph import StateGraph, START, END

class graph_schema(TypedDict):
    topic: str
    ig: str
    x: str
    li: str

demo_query: graph_schema = {
    "topic": "a very concise description of the PC game Unreal Tournament from 1999 by the campany Epic",
    "ig": "",
    "x": "",
    "li": "",
}

client = ChatOpenAI(
    model="qwen3.8:latest",
    temperature=0.6,
    top_p=0.95,
    base_url="http://localhost:11434/v1",
    api_key=SecretStr("ollama"),
)

def create_post_ig(state: graph_schema) -> dict:

    topic = state["topic"]
    post = client.invoke(f"Write a LinkedIn post about {topic}. Keep the tone Gen-z casual and engaging").content
    state["ig"] = str(post)

    return {"ig": post}

def create_post_x(state: graph_schema) -> dict:

    topic = state["topic"]
    post = client.invoke(f"Write a LinkedIn post about {topic}. Keep the tone like a mid-nineties action movie.").content
    state["x"] = str(post)

    return {"x": post}

def create_post_li(state: graph_schema) -> dict:

    topic = state["topic"]
    post = client.invoke(f"Write a LinkedIn post about {topic}. Keep the tone smart and business-like").content
    state["li"] = str(post)

    return {"li": post}

def build_state_graph():
    graph = StateGraph(graph_schema)

    graph.add_node("create_post_ig", create_post_ig)
    graph.add_node("create_post_x", create_post_x)
    graph.add_node("create_post_li", create_post_li)

    graph.add_edge(START, "create_post_ig")
    graph.add_edge(START, "create_post_x")
    graph.add_edge(START, "create_post_li")

    return graph.compile()


if __name__ == '__main__':
    parallel_graph = build_state_graph()
    response = parallel_graph.invoke(demo_query)

    print(response["ig"])
    print(response["x"])
    print(response["li"])

Workflow :: Routing​

Decision models - models that return a simple structured response instead of a wall of text - just became very popular. But the idea is not new ~ your regular model can do that as well... only much slower. So let's take the example from above and use our model in a schema-restrained form to decide what platform a post should land on based on it's topic - instead of writing all three:

from pydantic import BaseModel, Field, SecretStr
from typing import Literal, TypedDict
from langchain_openai import ChatOpenAI
from langgraph.graph import StateGraph, START, END

class model_schema(BaseModel):
    target: Literal['ig', 'x', 'li'] = Field(..., description="The best target social network to which to generate the post for based on the user query - Instagram, Twitter/X or LinkedIn")
    topic: str = Field(..., description="The topic for the post to generate.")

class graph_schema(TypedDict):
    target: str
    input: str
    topic: str
    post: str

client = ChatOpenAI(
    model="qwen3.8:latest",
    temperature=0.6,
    top_p=0.95,
    base_url="http://localhost:11434/v1",
    api_key=SecretStr("ollama"),
)

client_with_structured_output = client.with_structured_output(model_schema)

def decider_node(state: graph_schema) -> graph_schema:
    user_query = state["input"]

    target_decision = client_with_structured_output.invoke(user_query)

    state["target"] = target_decision.target
    state["topic"] = target_decision.topic

    return state

def condition(state: graph_schema) -> str:
    target = state["target"]

    if target == "ig":
        return 'trigger_create_post_ig'
    elif target == "x":
        return 'trigger_create_post_x'
    elif target == "li":
        return 'trigger_create_post_li'
    else:
        raise ValueError("Invalid target")

def create_post_ig(state: graph_schema) -> dict:

    input = state["input"]
    post = client.invoke(f"Write a LinkedIn post about {input}. Keep the tone Gen-z casual and engaging").content
    state["post"] = str(post)

    return {"post": post}

def create_post_x(state: graph_schema) -> dict:

    input = state["input"]
    post = client.invoke(f"Write a LinkedIn post about {input}. Keep the tone like a mid-nineties action movie.").content
    state["post"] = str(post)

    return {"post": post}

def create_post_li(state: graph_schema) -> dict:

    input = state["input"]
    post = client.invoke(f"Write a LinkedIn post about {input}. Keep the tone smart and business-like").content
    state["post"] = str(post)

    return {"post": post}

def build_state_graph():
    routed_graph = StateGraph(graph_schema)

    routed_graph.add_node("decider_node", decider_node)
    routed_graph.add_node("create_post_ig", create_post_ig)
    routed_graph.add_node("create_post_x", create_post_x)
    routed_graph.add_node("create_post_li", create_post_li)

    routed_graph.add_edge(START, "decider_node")
    routed_graph.add_conditional_edges("decider_node", condition, {
        "trigger_create_post_ig": "create_post_ig",
        "trigger_create_post_x": "create_post_x",
        "trigger_create_post_li": "create_post_li"
    })
    routed_graph.add_edge("create_post_ig", END)
    routed_graph.add_edge("create_post_x", END)
    routed_graph.add_edge("create_post_li", END)

    return routed_graph.compile()


if __name__ == '__main__':
    parallel_graph = build_state_graph()
    response = parallel_graph.invoke({
        "input": "Write a post about how sick 90s and early 2000s sitcoms were - contrast the differences between Love and a Mariage and the show Friends",
        "target": "",
        "topic": "",
        "post": ""
    })

    print(response["topic"])
    print(response["post"])

Workflow :: Orchestrator-Worker​

Let one orchestrator node delegate it's workload onto worker threads:

from pydantic import BaseModel, Field, SecretStr
from typing import TypedDict
from concurrent.futures import ThreadPoolExecutor

from langchain_core.prompts import ChatPromptTemplate
from langchain_openai import ChatOpenAI
from langgraph.graph import StateGraph, START, END


class model_schema(BaseModel):
    tasks: list[str] = Field(..., description="Evaluate the user query to generate this list of invidiual tasks to be performed by the worker.")

class graph_schema(TypedDict):
    tasks: list[str]
    query: str
    results: list[str]
    summary: str

client = ChatOpenAI(
    model="qwen3.8:latest",
    temperature=0.6,
    top_p=0.95,
    base_url="http://localhost:11434/v1",
    api_key=SecretStr("ollama"),
)

client_with_structured_output = client.with_structured_output(model_schema)


def orchestrator_node(state: graph_schema) -> graph_schema:
    user_query = state["query"]

    prompt = ChatPromptTemplate.from_messages([
        ("system", "You are an orchestrator that breaks down a user query into tasks for the worker."),
        ("user", f"User query: {user_query}. Please generate one prompt per task for the worker to complete.")
    ])

    chain = prompt | client_with_structured_output

    task_list = chain.invoke({"query": user_query})

    state["tasks"] = task_list.tasks

    return state


def execute(query: str):
    response = client.invoke(f"Please execute this task: {query}")

    return response.content


def worker_node(state: graph_schema) -> graph_schema:

    tasks = state["tasks"]
    results = []

    # spawn as many workers as there are tasks
    with ThreadPoolExecutor(max_workers=len(tasks)) as executer:
        # and divide the lists of tasks on your workers
        results_futures = (executer.map(execute, tasks))
        # once resolved append to results
        for result in results_futures:
            results.append(result)

    state["results"] = results

    return state


def collector_node(state: graph_schema):

    results = state["results"]

    prompt = ChatPromptTemplate.from_messages([
        ("system", "You are a collector that summarized the results from the worker."),
        ("user", f"Here are the results from the worker: {results}. Please provide an easy to read, concise and structured summary of the results.")
    ])

    chain = prompt | client

    summary = chain.invoke({"results": results})

    state["summary"] = str(summary.content)

    return state

def build_state_graph():
    orchestrator_graph = StateGraph(graph_schema)

    orchestrator_graph.add_node("orchestrator_node", orchestrator_node)
    orchestrator_graph.add_node("worker_node", worker_node)
    orchestrator_graph.add_node("collector_node", collector_node)

    orchestrator_graph.add_edge(START, "orchestrator_node")
    orchestrator_graph.add_edge("orchestrator_node", "worker_node")
    orchestrator_graph.add_edge("worker_node", "collector_node")
    orchestrator_graph.add_edge("collector_node", END)

    return orchestrator_graph.compile()


if __name__ == '__main__':
    orchestrator_graph = build_state_graph()
    response = orchestrator_graph.invoke({
        "query": "Give me the population numbers of the following cities: New York, Shanghai, Sydney, Hongkong, Berlin",
        "tasks": [],
        "results": [],
        "summary": ""
    })

    print(response["tasks"])
    print(response["summary"])

Workflow :: Generator-Evaluator​

Next - let's let an evaluator node critique the output of a generator node until a standard of quality is reached:

from pydantic import BaseModel, Field, SecretStr
from typing import TypedDict
from concurrent.futures import ThreadPoolExecutor

from langchain_core.prompts import ChatPromptTemplate
from langchain_openai import ChatOpenAI
from langgraph.graph import StateGraph, START, END


class model_schema(BaseModel):
    funny: bool = Field(..., description="True if the generated joke is funny, False if it is not.")
    feedback: str = Field(default="", description="Concrete feedback on how to make the joke funnier, or an empty string if it is already funny.")

class graph_schema(TypedDict):
    topic: str
    joke: str
    funny: bool
    feedback: str
    iterations: int

client = ChatOpenAI(
    model="qwen3.8:latest",
    temperature=0.6,
    top_p=0.95,
    base_url="http://localhost:11434/v1",
    api_key=SecretStr("ollama"),
)

client_with_structured_output = client.with_structured_output(model_schema)


def generator_node(state: graph_schema) -> graph_schema:

    topic = state["topic"]

    if state['feedback']:
        response = client.invoke(f"Please modify the following joke: {state['joke']} based on the evaluators feedback: {state['feedback']}")
    else:
        prompt = ChatPromptTemplate.from_messages([
            ("system", "You are a joke creator and will be evaluated on your joke quality. You will receive a topic and then generate a joke around it."),
            ("user", f"Here is the topic: {topic}. Please create a funny joke about it.")
        ])

        chain = prompt | client
        response = chain.invoke({"topic": topic})

    state["joke"] = str(response.content)

    return state

def evaluator_node(state: graph_schema) -> graph_schema:

    joke = state["joke"]
    iteration = state['iterations']
    prompt = ChatPromptTemplate.from_messages([
        ("system", "You are a comedy critic. Your job is to evaluat the following joke and provide feedback on how to make it funnier."),
        ("user", f"Evaluate the following joke: {joke}. Respond with True if it is funny and False if it is not and provide feedback if it's not funny.")
    ])

    chain = prompt | client_with_structured_output
    evaluation = chain.invoke({"joke": joke})
    state["funny"] = evaluation.funny
    state["feedback"] = evaluation.feedback
    state["iterations"] = iteration + 1

    return state

def stop(state: graph_schema) -> str:
    if state["funny"] or state["iterations"] >=5:
        return "end"
    else:
        return "evaluator_node"


def build_state_graph():
    evaluator_graph = StateGraph(graph_schema)

    evaluator_graph.add_node("generator_node", generator_node)
    evaluator_graph.add_node("evaluator_node", evaluator_node)

    evaluator_graph.add_edge(START, "generator_node")
    evaluator_graph.add_conditional_edges("generator_node", stop, {
        "evaluator_node": "evaluator_node",
        "end": END
    })
    evaluator_graph.add_edge("evaluator_node", "generator_node")
    evaluator_graph.add_edge("generator_node", END)

    return evaluator_graph.compile()


if __name__ == '__main__':
    evaluator_graph = build_state_graph()
    response = evaluator_graph.invoke({
        "topic": "The Rust programming language",
        "joke": "",
        "funny": False,
        "feedback": "",
        "iterations": 0,
    })

    print(response["funny"])
    print(response["joke"])

Memory​

For Chatbot interactions it is nice to have the agent remember the conversation in between prompts. This can be done manually by appending every conversation to the following prompt. But LangGraph also adds a CheckPoint function that does it for you:

from typing import Annotated, TypedDict
from langgraph.graph.message import add_messages
from pydantic import SecretStr
from langchain_openai import ChatOpenAI

from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import InMemorySaver

client = ChatOpenAI(
    model="qwen3.8:latest",
    temperature=0.6,
    top_p=0.95,
    base_url="http://localhost:11434/v1",
    api_key=SecretStr("ollama")
)

# Define the schema

class graph_shema(TypedDict):
    messages: Annotated[list[str], add_messages] # <- HAS TO BE CALLED MESSAGES, otherwise memory does not work...

# Create Node functions

def welcome(state: graph_shema) -> graph_shema:
    current_message = state['messages']

    resonse = client.invoke(current_message).content

    state['messages'] = [f"You asked {current_message}. Here is the response: {resonse}."]

    return state

# Create the State Graph with Memory Checkpoint

memory_graph = StateGraph(graph_shema)
checkpoint = InMemorySaver() # <- initialize the CheckPoint system

memory_graph.add_node("welcome", welcome)

memory_graph.add_edge(START, "welcome")
memory_graph.add_edge("welcome", END)
graph = memory_graph.compile(checkpointer=checkpoint) # <- Compile the graph with CheckPoints

# Run Graph with configuration

response = graph.invoke(
    {"messages": ["Hey there! My name is Mike. Who are you?"]},
    {"configurable":  {"thread_id": "this-is-a-memory-test"}} # <- Make sure that all requests with shared memory use the same thread ID
)

for message in response['messages']:
    message.pretty_print()

response2 = graph.invoke(
    {"messages": ["Hey there! Do you know who I am?"]},
    {"configurable":  {"thread_id": "this-is-a-memory-test"}} # <- Make sure that all requests with shared memory use the same thread ID
)

for message in response2['messages']:
    message.pretty_print()

Human in the Loop​

Add a node that forces the agent to seek user approval before continuing with a task. To avoid having to re-run the entire graph this feature is build on the Memory feature above - allowing the agent to resume operation after the approval was received.

from typing import Annotated, Literal, Optional, TypedDict
from langgraph.graph.message import add_messages
from pydantic import SecretStr
from langchain_openai import ChatOpenAI

from langgraph.graph import StateGraph, START, END
from langgraph.types import Command, interrupt
from langgraph.checkpoint.memory import MemorySaver

client = ChatOpenAI(
    model="qwen3.8:latest",
    temperature=0.6,
    top_p=0.95,
    base_url="http://localhost:11434/v1",
    api_key=SecretStr("ollama")
)

# Define the schema

class graph_shema(TypedDict):
    messages: Annotated[list[str], add_messages] # <- HAS TO BE CALLED MESSAGES, otherwise memory does not work...

class ApprovalState(TypedDict):
    action_details: str
    status: Optional[Literal["pending", "approved", "rejected"]]

# Create Node functions

def approval_node(state: ApprovalState) -> Command[Literal["proceed", "cancel"]]:
   decision = interrupt({
   "question": "Do you approve this action?",
   "details": state["action_details"],
   })

   return Command(goto="proceed" if decision else "cancel")

def proceed_node(state: ApprovalState):
    return {"status": "approved"}

def cancel_node(state: ApprovalState):
    return {"status": "rejected"}

approval_graph = StateGraph(ApprovalState)
checkpoint = MemorySaver()

approval_graph.add_node("approval", approval_node)
approval_graph.add_node("proceed", proceed_node)
approval_graph.add_node("cancel", cancel_node)

approval_graph.add_edge(START, "approval")
approval_graph.add_edge("proceed", END)
approval_graph.add_edge("cancel", END)

graph = approval_graph.compile(checkpointer=checkpoint)
config = {"configurable": {"thread_id": "approval-flow"}}

# Run Graph with configuration

initial = graph.invoke(
    {"action_details": "Send sumary by Email?", "status": "pending"},
    config=config,
)

# The graph is now PAUSED at the interrupt: nothing after "approval" has run.
# The pending question lives in state, keyed by "__interrupt__":
pending = initial["__interrupt__"][0].value
print(pending)  # {'question': 'Do you approve this action?', 'details': 'Send sumary by Email?'}

# --- THIS is the human in the loop: stop the process and wait for a real answer.
answer = input(f"{pending['question']} ({pending['details']}) [y/n] ").strip().lower()
resume_value = answer in ("y", "yes")

# Resuming on the SAME thread_id replays the interrupted node;
# interrupt() returns whatever we resume with, so the branch follows the answer.
resumed = graph.invoke(
    Command(resume=resume_value),
    config=config,
)

print(resumed["status"])
